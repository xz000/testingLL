/// 由「第一名奖励」自动生成名次奖励档位：每降一名奖励 ×0.6（向下取整），直到 ≤0。
/// 这样只需输一个数字即可覆盖任意玩家数（档位只影响前几名，后几名逐渐归零）。
#[cfg(feature = "steam")]
fn auto_place_rewards(first: i32) -> Vec<i32> {
    let mut out = Vec::new();
    let mut v = first.max(0);
    while v > 0 {
        out.push(v);
        if out.len() >= 64 {
            break;
        }
        v = (v as f64 * 0.6).floor() as i32;
    }
    if out.is_empty() {
        out.push(0);
    }
    out
}
