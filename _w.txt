958     fn shrink_arena(&mut self, dt: Fix64) {
959         let alive = self.players.iter().filter(|p| p.alive).count().max(1) as f64;
960         if self.shrink_timer > Fix64::ZERO {
961             self.shrink_timer -= dt;
962             return;
963         }
964         let b = Balance::default();
965         let rate = b.ring_width / (b.shrink_ring_secs * alive.sqrt());
966         self.arena_radius = (self.arena_radius - Fix64::from_num(rate * dt.to_num::<f64>())).max(Fix64::ZERO);
967     }
