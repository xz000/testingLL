==== handler KC ====

function KC takes nothing returns nothing
local integer dX=Kv
local integer HO=0
local real NO
local integer n=0
local real DX
local boolean lC=rV[$C+24*ai]
if rV[$D+24*ai]then
call kC(ai,ni)
endif
if ei[ai]==0 then
set DX=$A
else
set DX=$B+ei[ai]
call DestroyEffect(AddSpecialEffectTarget("Abilities\\Spells\\Human\\Invisibility\\InvisibilityTarget.mdl",Xn[ai],"right hand"))
endif
if lC then
set xi[ai]=2*xi[ai]
endif
if aV[ai]and GetTerrainType(K[ni],L[ni])==sa[ka]then
call HC(ai,ni)
else
call CX(ni,DX)
endif
loop
exitwhen dX==0
set HO=rv[dX]
if dn[Vv[dX]]!=dn[ai]then
set NO=uX(dX,ni)
if NO<=$FA then
call KI(ni,dX,DX,1.-NO/ $3E8)
set n=n+1
endif
if Fv[dX]==ni and(GetTerrainType(K[dX],L[dX])==sa[ka]or lr[Vv[dX]])then
call oR(dX)
if GetLocalPlayer()==Player(Vv[ni])or GetLocalPlayer()==Player(Vv[dX])then
call StartSound(Wx)
set rr=CreateTextTag()
call SetTextTagPos(rr,K[ni],L[ni],0)
call SetTextTagPermanent(rr,false)
call SetTextTagColor(rr,$FF,50,50,$DC)
call SetTextTagText(rr,"否决",.025)
call SetTextTagLifespan(rr,1.5)
call SetTextTagFadepoint(rr,1.)
call SetTextTagVelocity(rr,0,.055)
endif
endif
endif
set dX=HO
endloop
if ei[ai]==2 then
call SetUnitState(F[ni],UNIT_STATE_MANA,GetUnitState(F[ni],UNIT_STATE_MANA)-4*n)
endif
if lC then
set xi[ai]=xi[ai]/ 2
endif
if n>=3 and not Tn then
set rr=CreateTextTag()
if lC then
call SetTextTagText(rr,"吸血鬼",.025)
call StartSound(Ro)
else
call SetTextTagText(rr,"帽子戏法",.025)
call StartSound(Ux)
endif
call SetTextTagPos(rr,K[ni],L[ni],0)
call SetTextTagPermanent(rr,false)
call SetTextTagColor(rr,50,$FF,$FF,$DC)
call SetTextTagLifespan(rr,1.5)
call SetTextTagFadepoint(rr,1.)
call SetTextTagVelocity(rr,0,.055)
call DestroyEffect(AddSpecialEffect("ArcaneExplosion.mdx",K[ni],L[ni]))
endif
if rV[17+24*ai]and Ja[ai]then
call gC(ai,ni,1)
call GX(ai)
return
endif
call DestroyEffect(AddSpecialEffect("Abilities\\Spells\\Demon\\DarkPortal\\DarkPortalTarget.mdl",K[ni],L[ni]))
call DestroyEffect(AddSpecialEffect("Nova_yellow.mdl",K[ni],L[ni]))

==== handler pb ====

function pb takes nothing returns nothing

set dy=dy/ NO

set jv[Ib]=Fi

==== handler ub ====

function ub takes nothing returns nothing
local integer Ib
local real dx
local real dy
local real NO
local real Ub=$5DC*.03
local real wb=300*.03
local real Wb
local real yb
local real Yb=800*(1+1.5*.1*oi[ai])
call kX()
set dx=TX-K[ni]
set dy=TY-L[ni]
set Ib=VO(2,ai,'e000',K[ni],L[ni],270,"ArcaneGlaive.mdl")
call SetUnitColor(F[Ib],GetPlayerColor(Player(ai)))
set Ev[Ib]=2
set Xv[Ib]=Wr[ai]
set Av[Ib*3+1]=true
set Av[Ib*3+3]=true
set Rv[Ib]=40
set hv[Ib]=Bi
set xv[Ib]=1
set Gv[Ib]=ci
set NO=SquareRoot(dx*dx+dy*dy)
if NO<300 then
set NO=300
endif
set dy=dy/ NO
set dx=dx/ NO
if NO>Yb then
set NO=Yb
endif
set Wb=-(Ub*Ub)/(2*NO)
set yb=2*Wb*wb/ Ub
if ja[ai]then
set wb=-wb
set yb=-yb
set ja[ai]=false
else
set ja[ai]=true
endif
set Q[Ib]=Ub*dx+wb*(-dy)
set S[Ib]=Ub*dy+wb*(dx)
set U[Ib]=Wb*dx+yb*(-dy)
set w[Ib]=Wb*dy+yb*(dx)
set ev[Ib]=-Ub/ Wb*.03
set gv[Ib]=ev[Ib]-5*.03
set Y[Ib]=Wb*dx-yb*(-dy)
set z[Ib]=Wb*dy-yb*dx
